	# In binary

	baseRule = baseRuleWhenOneMeetOne()

def mergeArrayOfBits(arrayA, arrayB):
	lenA = len(arrayA)
	lenB = len(arrayB)
	if (lenA < lenB) :
		tmp = arrayA
		arrayA = arrayB
		arrayB = tmp
		lenA = len(arrayA)
		lenB = len(arrayB)
